cmake_minimum_required(VERSION 3.16)
project(vecsearch LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native -DNDEBUG")
find_package(OpenMP)
find_package(Threads REQUIRED)
add_library(vs INTERFACE)
target_include_directories(vs INTERFACE include)
target_link_libraries(vs INTERFACE Threads::Threads)
foreach(t bench bench_filtered bench_stream)
  add_executable(${t} bench/${t}.cpp)
  target_link_libraries(${t} vs)
  if(OpenMP_CXX_FOUND)
    target_link_libraries(${t} OpenMP::OpenMP_CXX)
  endif()
endforeach()
enable_testing()
add_executable(test_hnsw tests/test_hnsw.cpp)
target_link_libraries(test_hnsw vs)
if(OpenMP_CXX_FOUND)
  target_link_libraries(test_hnsw OpenMP::OpenMP_CXX)
endif()
add_test(NAME hnsw COMMAND test_hnsw)
